#!/usr/bin/env python3
"""Run each old G2 QA file (taken from its own PR HEAD) on G1 #468 and on G1 #497, in a throwaway worktree."""
import json, os, re, subprocess, sys

WT = sys.argv[1]
NM = '<node_modules of a G2 worktree>'
HEADS = {'468': '45ee9ac3296fdc6f5e53942faa93a1e73e61c1c9', '497': '446a9c3960262a078aa7784503673e87edbae331'}
OLD = [
    ('394', '7384aa5abb29496c01d63c2d0add2cce5dcddc2b', 'g2-x-manual-result-gate.test.mjs'),
    ('403', 'ffed4d8735c5b270c87679f259cdc3aa30007f2c', 'g2-pr398-regression.test.mjs'),
    ('411', '38501193d7baf921f673e5e169e69752f34eae3e', 'g2-postfix-baseline.test.mjs'),
    ('414', '7b150a89d52bae0692487654fa4c1ae7db398112', 'g2-pr410-transitions.test.mjs'),
    ('448', 'd44713bc84407847aee66a131c3c67adad6fc320', 'g2-pr435-retry-kind.test.mjs'),
]
APP = os.path.join(WT, 'apps/social-mobile')
LINE = re.compile(r'^(✔|✖|﹣) (.*?)(?: \(\d+(?:\.\d+)?m?s\))?(?: # (TODO|SKIP).*)?$')

def run(cmd, **kw):
    return subprocess.run(cmd, capture_output=True, text=True, **kw)

out = {}
for head, sha in HEADS.items():
    assert run(['git', '-C', WT, 'checkout', '-q', '--detach', '--force', sha]).returncode == 0
    link = os.path.join(APP, 'node_modules')
    if not os.path.lexists(link):
        os.symlink(NM, link)
    for pr, prsha, name in OLD:
        body = run(['git', '-C', WT, 'show', f'{prsha}:apps/social-mobile/tests/{name}']).stdout
        dest = os.path.join(APP, 'tests', name)
        open(dest, 'w', encoding='utf-8').write(body)
        r = run(['node', '--test', '--test-timeout=60000', f'tests/{name}'], cwd=APP)
        res = {}
        for ln in (r.stdout + r.stderr).splitlines():
            m = LINE.match(ln)
            if m and m.group(2) != 'failing tests:' and m.group(2) not in res:
                res[m.group(2)] = 'TODO' if m.group(3) == 'TODO' else ('PASS' if m.group(1) == '✔' else 'FAIL')
        summ = dict(re.findall(r'^ℹ (tests|pass|fail|todo|skipped) (\d+)$', r.stdout + r.stderr, re.M))
        out.setdefault(pr, {'pr_head': prsha, 'file': name})[head] = {'summary': summ, 'results': res}
        os.remove(dest)
    os.remove(link)
    run(['git', '-C', WT, 'checkout', '-q', '--force', '--', '.'])
json.dump(out, sys.stdout, ensure_ascii=False, indent=1)
